% ==============================================================================
% PAQUETES Y CONFIGURACIÓN GENERAL (APA 7)

% Figuras y tablas se muestran en línea (no al final): se usa la opción de
% clase "floatsintext" (ver main.tex), que evita que apa7 cargue endfloat.

\usepackage[utf8]{inputenc}
\usepackage[spanish, es-tabla]{babel} % es-tabla asegura "Tabla" en lugar de "Cuadro"
\usepackage{csquotes}
\usepackage{graphicx}
\usepackage{float}                 % Habilita el especificador [H] para fijar figuras en su lugar exacto
\usepackage{pgfplots}              % Gráficos nativos (barras, tortas, líneas)
\pgfplotsset{compat=1.18}
\usetikzlibrary{positioning,arrows.meta,fit,backgrounds}
\usepackage{pgf-pie}               % Gráfico de torta/pastel
\usepackage{pgfgantt}              % Diagramas de Gantt (cronograma)
\usepackage{etoolbox}              % \patchcmd (parche de bandas de pgfgantt)

% Parche de pgfgantt: inyecta \GanttBands dentro de \endganttchart para poder
% dibujar bandas de color de fondo antes de la rejilla. La redefinición concreta
% de \GanttBands se realiza en la sección del cronograma (08_cronograma.tex).
\makeatletter
\newcommand\GanttBands{}%
\patchcmd{\endganttchart}{\ifgtt@vgrid}{\GanttBands\ifgtt@vgrid}{}{%
  \PackageError{}{No se pudo parchear pgfgantt para las bandas}{}%
}
\makeatother

\usepackage{booktabs}  % Tablas elegantes según estándar APA (sin líneas verticales)
\usepackage{array}     % Columnas con formato avanzado (p{}, >{}, \arraybackslash)
\usepackage{float}     % Posicionamiento exacto de tablas y figuras ([H])
\usepackage{amsmath}
\usepackage{ragged2e} % Permite forzar justificación plena (\justifying)
\usepackage{indentfirst} % Sangría de primera línea en TODO párrafo, incluido el primero tras cada encabezado
\usepackage{microtype} % Ajuste fino de espaciado/expansión de letras (estilo Word) para justificar sin desbordes
\usepackage{hyperref}

% Estilo "Word": justificado sin cortar palabras al final de línea.
% Se desactiva el silabeo y se permite estirar/kern los espacios entre
% palabras (como hace Microsoft Word) en lugar de dividir con guion.
\hyphenpenalty=10000
\exhyphenpenalty=10000
\tolerance=9999
\emergencystretch=6em

% Configuración de citas y bibliografía APA 7.ª Edición
\usepackage[style=apa, backend=biber]{biblatex}
\addbibresource{referencias.bib}

% Enlaces configurados en color negro para un aspecto tradicional impreso.
% Los enlaces siguen siendo funcionales en el PDF, pero sin tinta de color
% (puntos de guía en el índice, sin resaltado azul).
\hypersetup{
    colorlinks=true,
    linkcolor=black,
    citecolor=black,
    urlcolor=black
}

% ==============================================================================
% ENCABEZADOS ADA 7 (sin running head del título del proyecto)
% La carátula (titlepage) no muestra número de página; las páginas interiores
% muestran únicamente el número en la esquina superior derecha.
% ==============================================================================
\fancypagestyle{titlepage}{%
    \fancyhf{}%
    \renewcommand{\headrulewidth}{0pt}%
}
\fancypagestyle{otherpage}{%
    \fancyhf{}%
    \rhead{\thepage}%
    \renewcommand{\headrulewidth}{0pt}%
}
\pagestyle{otherpage}

% ==============================================================================
% PROFUNDIDAD DE NUMERACIÓN Y TABLA DE CONTENIDOS (ÍNDICE)
% secnumdepth=0 evita duplicar la numeración manual de los títulos en el índice.
% tocdepth=5 permite incluir hasta el nivel 5 (5.1.1.1. Inicio, etc.).
% ==============================================================================
\setcounter{secnumdepth}{0}
\setcounter{tocdepth}{5}

% Corrección para que \subparagraph registre su título en el TOC bajo apa7
\makeatletter
\renewcommand{\subparagraph}[1]{%
  \@startsection{subparagraph}{5}{\parindent}{\b@level@five@skip}{\e@level@five@skip}%
  {\normalfont\normalsize\bfseries\itshape}{#1}%
}
\makeatother

% Líneas punteadas (puntos de guía) en el Índice General para los títulos de
% nivel 1. El article.cls actual define \l@section sin puntos; se restaura el
% formato clásico con \@dottedtocline conservando la alineación a la izquierda.
\makeatletter
\renewcommand*\l@section{\@dottedtocline{1}{0em}{2.3em}}
\makeatother

% ==============================================================================
% ÍNDICE DE TABLAS Y FIGURAS: formato "Tabla 1. ..." / "Figura 1. ..."
% ==============================================================================
\makeatletter
% Índice de Tablas: anteponer "Tabla" al número de tabla
\renewcommand*\l@table[2]{%
  \begingroup
    \renewcommand*\numberline[1]{\hb@xt@\@tempdima{\textbf{Tabla~##1.}~\hfil}}%
    \@dottedtocline{1}{0em}{6.4em}{#1}{#2}%
  \endgroup
}
% Índice de Figuras: anteponer "Figura" al número de figura
\renewcommand*\l@figure[2]{%
  \begingroup
    \renewcommand*\numberline[1]{\hb@xt@\@tempdima{\textbf{Figura~##1.}~\hfil}}%
    \@dottedtocline{1}{0em}{6.4em}{#1}{#2}%
  \endgroup
}
\makeatother
