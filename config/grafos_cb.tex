%                cgf = final «F» (círculo doble), cgi = «Inicio» (óvalo)
%   Aristas:  \draw[fca] (gA) -- node[right,font=\scriptsize]{F} (gB);  (flecha con etiqueta V/F)

\newcommand{\cgn}[5]{\node[#4] (g#1) at (#2,{-#3*1.25}) {#5};}
\tikzset{
    cgs/.style={circle, draw, minimum size=8.5mm, inner sep=0pt, font=\footnotesize},
    cgp/.style={cgs, fill=gray!30},
    cgf/.style={cgs, double},
    cgi/.style={rectangle, rounded corners=6pt, draw, minimum height=6mm, minimum width=13mm, inner sep=2pt, font=\footnotesize},
    fca/.style={-{Stealth[length=2mm]}, semithick}
}
